`app/registry.py`:
```python
from __future__ import annotations
from dataclasses import dataclass, field
from pathlib import Path
import yaml

class RegistryError(ValueError):
    pass

@dataclass
class InjectPoint:
    node: str
    field: str

@dataclass
class ModelEntry:
    name: str
    template: str
    available: bool = True
    duration_range: list = field(default_factory=lambda: [3, 8])
    fps: int = 25
    resolutions: dict = field(default_factory=lambda: {"720p": 720})
    dims_multiple: int = 32
    aspects: list = field(default_factory=lambda: ["9:16", "16:9", "1:1"])
    timeout_s: int = 1800
    inject: dict = field(default_factory=dict)  # param -> list[InjectPoint]

    def validate(self, duration: int, resolution: str) -> tuple[int, int]:
        lo, hi = self.duration_range
        if not (lo <= int(duration) <= hi):
            raise RegistryError(f"duration {duration} 超出 [{lo},{hi}]")
        if resolution not in self.resolutions:
            raise RegistryError(f"resolution 不支持: {resolution}，可选 {list(self.resolutions)}")
        return int(duration), self.resolutions[resolution]

@dataclass
class Registry:
    entries: dict

    def get_entry(self, name: str) -> ModelEntry:
        if name not in self.entries:
            raise RegistryError(f"未注册模型: {name}")
        return self.entries[name]

def load_registry(path: Path) -> Registry:
    entries: dict[str, ModelEntry] = {}
    for raw in yaml.safe_load(Path(path).read_text(encoding="utf-8")) or []:
        inj = {k: [InjectPoint(**p) for p in v] for k, v in (raw.get("inject") or {}).items()}
        e = ModelEntry(name=raw["name"], template=raw["template"],
                       available=raw.get("available", True),
                       duration_range=raw.get("duration_range", [3, 8]),
                       fps=raw.get("fps", 25),
                       resolutions=raw.get("resolutions", {"720p": 720}),
                       dims_multiple=raw.get("dims_multiple", 32),
                       aspects=raw.get("aspects", ["9:16", "16:9", "1:1"]),
                       timeout_s=raw.get("timeout_s", 1800), inject=inj)
        entries[e.name] = e
    return Registry(entries)
```

`app/task_store.py`:
```python
from __future__ import annotations
import json, sqlite3, time
from pathlib import Path

_SCHEMA = """
CREATE TABLE IF NOT EXISTS tasks(
  id TEXT PRIMARY KEY, prompt_id TEXT, model TEXT, status TEXT,
  params TEXT, video_file TEXT DEFAULT '', error TEXT DEFAULT '',
  created_at REAL, updated_at REAL);
"""

class TaskStore:
    def __init__(self, db_path: Path):
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self._conn = sqlite3.connect(self.db_path, check_same_thread=False)
        self._conn.row_factory = sqlite3.Row

    def init(self):
        self._conn.execute(_SCHEMA); self._conn.commit()

    def create(self, task_id: str, prompt_id: str, model: str, params: dict):
        now = time.time()
        self._conn.execute(
            "INSERT INTO tasks VALUES(?,?,?,?,?,?,?,?,?)",
            (task_id, prompt_id, model, "queued", json.dumps(params), "", "", now, now))
        self._conn.commit()

    def get(self, task_id: str) -> dict | None:
        r = self._conn.execute("SELECT * FROM tasks WHERE id=?", (task_id,)).fetchone()
        return dict(r) if r else None

    def set_status(self, task_id: str, status: str, error: str = "", video_file: str = ""):
        self._conn.execute(
            "UPDATE tasks SET status=?, error=?, video_file=?, updated_at=? WHERE id=?",
            (status, error, video_file, time.time(), task_id))
        self._conn.commit()

    def fail_pending(self, reason: str):
        self._conn.execute(
            "UPDATE tasks SET status='failed', error=?, updated_at=? "
            "WHERE status IN ('queued','running')", (reason, time.time()))
        self._conn.commit()
```

`app/templating.py`:
```python
from __future__ import annotations
import copy
from app.registry import ModelEntry

def dims_for(aspect_wh: tuple[int, int], short_side: int, multiple: int) -> tuple[int, int]:
    """按首帧长宽比与短边档位求宽高，向上对齐到 multiple 倍数。"""
    w0, h0 = aspect_wh
    if w0 >= h0:   # 横版或方图：短边=高
        h = short_side; w = round(w0 / h0 * short_side)
    else:
        w = short_side; h = round(h0 / w0 * short_side)
    up = lambda v: ((v + multiple - 1) // multiple) * multiple
    return up(w), up(h)

def frames_for(duration_s: int, fps: int) -> int:
    return duration_s * fps + 1

def build_graph(entry: ModelEntry, template: dict, image_name: str, prompt: str,
                width: int, height: int, length: int, fps: int | None = None) -> dict:
    g = copy.deepcopy(template)
    vals = {"image": image_name, "prompt": prompt, "width": width, "height": height,
            "length": length, "fps": fps}
    for param, points in entry.inject.items():
        v = vals.get(param)
        if v is None:
            continue
        for p in points:
            g[p.node]["inputs"][p.field] = v
    return g
```

`scripts/ui_to_api.py`:
```python
"""UI 格式工作流 → API 格式 prompt graph。支持子图（definitions.subgraphs）展平：
子图节点 id 重写为 "<instance_id>.<inner_id>"，接口链接按 sg.inputs/outputs 的 linkIds 穿透重接。
widget 值按 object_info 的 required+optional 顺序对齐到 inputs 字段名。
用法: python scripts/ui_to_api.py <ui.json> <out.json> <object_info.json>
"""
from __future__ import annotations
import copy, json, sys
from pathlib import Path

SKIP_TYPES = {"MarkdownNote", "Note"}

def _widget_names(cls: str, oi: dict) -> list[str]:
    """按 object_info 顺序返回该类的 widget 型输入名（connected-only 型除外）。"""
    spec = oi.get(cls, {}).get("input", {})
    names = []
    for group in ("required", "optional"):
        for name, typ in (spec.get(group) or {}).items():
            t0 = typ[0] if isinstance(typ, list) and typ else typ
            if isinstance(t0, list):      # combo（下拉）→ widget
                names.append(name)
            elif isinstance(t0, str) and t0 in ("INT", "FLOAT", "STRING", "BOOLEAN", "COMBO"):
                names.append(name)
    return names

def _node(n: dict, oi: dict) -> dict:
    """单节点 → {class_type, inputs}。widget 值按顺序填入，连接留给 links 阶段覆盖。"""
    inputs: dict = {}
    wv = list(n.get("widgets_values") or [])
    wi = 0
    for name in _widget_names(n["type"], oi):
        if wi < len(wv):
            inputs[name] = wv[wi]; wi += 1
    return {"class_type": n["type"], "inputs": inputs}

def _slot_out_name(cls: str, slot: int, oi: dict) -> str:
    outs = oi.get(cls, {}).get("output") or []
    return str(outs[slot]) if slot < len(outs) else str(slot)

def convert(ui: dict, oi: dict) -> dict:
    graph: dict = {}
    sgs = {sg["id"]: sg for sg in ui.get("definitions", {}).get("subgraphs", [])}
    ui_links = {l[0]: l for l in ui.get("links", [])}

    def outer_link_for(target_id: int, target_slot: int):
        for l in ui.get("links", []):
            if l[3] == target_id and l[4] == target_slot:
                return l
        return None

    # 1) 普通外层节点
    for n in ui["nodes"]:
        if n["type"] in SKIP_TYPES:
            continue
        if n["type"] in sgs:
            continue    # 子图实例在 2) 处理
        graph[str(n["id"])] = _node(n, oi)

    # 2) 子图实例展平
    for n in ui["nodes"]:
        sg = sgs.get(n["type"])
        if sg is None:
            continue
        prefix = str(n["id"])
        inner_links = {l[0]: l for l in sg.get("links", [])}
        for inner in sg["nodes"]:
            if inner["type"] in SKIP_TYPES:
                continue
            graph[f"{prefix}.{inner['id']}"] = _node(inner, oi)
        # 2a) 实例的 widget 输入（如 prompt STRING）：值直接写进 inputNode 下游
        #     实例 widgets_values 与 sg.inputs 中带 widget 的项按序对应
        widget_inputs = [i for i in n.get("inputs", []) if i.get("widget") and i.get("link") is None]
        wv = list(n.get("widgets_values") or [])
        for idx, wi in enumerate(widget_inputs):
            if idx >= len(wv):
                break
            val = wv[idx]
            # 找子图内从 inputNode 该 widget 位发出的内部链接，把值写到消费节点
            pos = [k for k, i2 in enumerate(sg.get("inputs", [])) if i2.get("widget")] or [idx]
            if idx < len(pos):
                for lid in (sg["inputs"][pos[idx]].get("linkIds") or []):
                    il = inner_links.get(lid)
                    if il:
                        tgt = graph.get(f"{prefix}.{il[3]}")
                        if tgt:
                            field = _first_free_str_field(tgt, val)
                            if field:
                                tgt["inputs"][field] = val
        # 2b) 连接输入（如 first_frame IMAGE）：外层源 → 子图内消费节点
        for si, sgin in enumerate(sg.get("inputs", [])):
            ol = outer_link_for(n["id"], si)
            if not ol:
                continue
            src_key, src_slot = str(ol[1]), ol[2]
            src_cls = graph[src_key]["class_type"]
            src_name = _slot_out_name(src_cls, src_slot, oi)
            for lid in (sgin.get("linkIds") or []):
                il = inner_links.get(lid)
                if il:
                    tgt = graph[f"{prefix}.{il[3]}"]
                    fname = _in_field_for(tgt, il[4], oi)
                    tgt["inputs"][fname] = [src_key, src_name]
        # 2c) 输出：子图内源 → 外层消费节点
        for so, sgout in enumerate(sg.get("outputs", [])):
            for lid in (sgout.get("linkIds") or []):
                il = inner_links.get(lid)
                if not il:
                    continue
                src_inner = graph[f"{prefix}.{il[1]}"]
                src_name = _slot_out_name(src_inner["class_type"], il[2], oi)
                for ol in ui.get("links", []):
                    if ol[1] == n["id"] and ol[2] == so:
                        tgt = graph.get(str(ol[3]))
                        if tgt:
                            fname = _in_field_for(tgt, ol[4], oi)
                            tgt["inputs"][fname] = [f"{prefix}.{il[1]}", src_name]

    # 3) 外层普通节点间连接
    for l in ui.get("links", []):
        src_id, src_slot, tgt_id, tgt_slot = l[1], l[2], l[3], l[4]
        if str(src_id) in sgs or str(tgt_id) in sgs or ui["nodes"] and any(
                n["id"] == tgt_id and n["type"] in sgs for n in ui["nodes"]):
            continue    # 子图相关连接已在 2) 处理
        s, t = graph.get(str(src_id)), graph.get(str(tgt_id))
        if not s or not t:
            continue
        t["inputs"][_in_field_for(t, tgt_slot, oi)] = [str(src_id), _slot_out_name(s["class_type"], src_slot, oi)]
    return graph

def _first_free_str_field(node: dict, val) -> str | None:
    for k, v in node["inputs"].items():
        if isinstance(v, str) and v == "" and isinstance(val, str):
            return k
    return None

def _in_field_for(tgt: dict, slot: int, oi: dict) -> str:
    """目标输入字段名：优先 object_info 可连接输入顺序，其次节点 inputs 数组名。"""
    cls = tgt["class_type"]
    spec = oi.get(cls, {}).get("input", {})
    conn = []
    for group in ("required", "optional"):
        for name, typ in (spec.get(group) or {}).items():
            t0 = typ[0] if isinstance(typ, list) and typ else typ
            if not (isinstance(t0, list) or (isinstance(t0, str) and t0 in
                    ("INT", "FLOAT", "STRING", "BOOLEAN", "COMBO"))):
                conn.append(name)   # 强类型（MODEL/IMAGE/...）= 连接型
    if slot < len(conn):
        return conn[slot]
    # 兜底：返回 inputs 中第一个值为空/None 的键
    for k, v in tgt["inputs"].items():
        if v is None or v == "":
            return k
    return list(tgt["inputs"])[0]

def main():
    src, dst, oi_path = sys.argv[1], sys.argv[2], sys.argv[3]
    ui = json.loads(Path(src).read_text(encoding="utf-8"))
    oi = json.loads(Path(oi_path).read_text(encoding="utf-8"))
    g = convert(ui, oi)
    Path(dst).parent.mkdir(parents=True, exist_ok=True)
    Path(dst).write_text(json.dumps(g, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"{len(g)} nodes -> {dst}")

if __name__ == "__main__":
    main()
```

from __future__ import annotations
import uuid
import httpx

class ComfySubmitError(RuntimeError):
    def __init__(self, msg: str, node_errors: dict | None = None):
        super().__init__(msg)
        self.node_errors = node_errors or {}

def parse_history(entry: dict) -> tuple[str, list[dict], str]:
    """history 条目 → (状态, 视频文件列表, 错误摘要)。completed 且有产物=succeeded；
    execution_error=failed；否则 running。"""
    if not entry:
        return "running", [], ""
    status = entry.get("status") or {}
    msgs = status.get("messages") or []
    err = ""
    for m in msgs:
        if isinstance(m, list) and m and m[0] == "execution_error":
            d = m[1] if len(m) > 1 else {}
            err = f"{d.get('node_type', '?')}: {d.get('exception_message', '')}"
    if err:
        return "failed", [], err
    vids: list[dict] = []
    for node_out in (entry.get("outputs") or {}).values():
        for key in ("videos", "gifs", "images"):
            for f in (node_out.get(key) or []):
                fn = f.get("filename", "")
                if fn.endswith((".mp4", ".webm")):
                    vids.append(f)
    if status.get("completed") and vids:
        return "succeeded", vids, ""
    if status.get("completed"):
        return "failed", [], "completed 但无视频输出"
    return "running", [], ""

class ComfyClient:
    def __init__(self, base_url: str, http: httpx.AsyncClient | None = None):
        self.base_url = base_url.rstrip("/")
        self.client_id = str(uuid.uuid4())
        self._own = http is None
        self.http = http or httpx.AsyncClient(timeout=30)

    async def aclose(self):
        if self._own:
            await self.http.aclose()

    async def upload_image(self, data: bytes, name: str) -> str:
        r = await self.http.post(f"{self.base_url}/upload/image",
                                 files={"image": (name, data, "image/png")})
        r.raise_for_status()
        return r.json()["name"]

    async def queue_prompt(self, graph: dict) -> str:
        r = await self.http.post(f"{self.base_url}/prompt",
                                 json={"prompt": graph, "client_id": self.client_id})
        if r.status_code != 200:
            try:
                d = r.json()
                raise ComfySubmitError(str(d.get("error"))[:300], d.get("node_errors"))
            except ValueError:
                raise ComfySubmitError(f"HTTP {r.status_code}: {r.text[:300]}")
        return r.json()["prompt_id"]

    async def history(self, prompt_id: str) -> dict | None:
        r = await self.http.get(f"{self.base_url}/history/{prompt_id}")
        r.raise_for_status()
        return (r.json() or {}).get(prompt_id)

    async def fetch_output(self, filename: str, subfolder: str = "", ftype: str = "output") -> bytes:
        r = await self.http.get(f"{self.base_url}/view",
                                params={"filename": filename, "subfolder": subfolder, "type": ftype})
        r.raise_for_status()
        return r.content

    async def object_info(self) -> dict:
        r = await self.http.get(f"{self.base_url}/object_info")
        r.raise_for_status()
        return r.json()
```

        raise HTTPException(400, detail=str(e))
    img_url = next((i.image_url.get("url", "") for i in body.content
                    if i.type == "image_url" and i.image_url), "")
    prompt = next((i.text for i in body.content if i.type == "text" and i.text), "")
    if not img_url or not prompt:
        raise HTTPException(400, detail="content 需含首帧图 data URL 与提示词文本")
    try:  # 解码首帧图，读宽高
        b64 = img_url.split(",", 1)[1]
        raw = base64.b64decode(b64)
        w, h = Image.open(io.BytesIO(raw)).size
    except Exception:
        raise HTTPException(400, detail="首帧图解码失败（需 data:image/...;base64）")
    template = Path(app.state.workflows_dir, entry.template)
    if not template.exists():
        raise HTTPException(500, detail=f"模板缺失: {entry.template}")
    tpl_json = json.loads(template.read_text(encoding="utf-8"))
    gw, gh = dims_for((w, h), short_side, entry.dims_multiple)
    tid = _new_task_id()
    comfy = app.state.comfy
    try:
        name = await comfy.upload_image(raw, tid + "_first.png")
        graph = build_graph(entry, tpl_json, image_name=name, prompt=prompt,
                            width=gw, height=gh,
                            length=frames_for(duration, entry.fps), fps=entry.fps)
        pid = await comfy.queue_prompt(graph)
    except ComfySubmitError as e:
        raise HTTPException(400, detail=f"ComfyUI 拒绝: {e}")
    except Exception as e:
        raise HTTPException(503, detail=f"ComfyUI 不可达: {e}")
    app.state.store.create(tid, pid, body.model,
                           {"duration": duration, "resolution": body.resolution,
                            "width": gw, "height": gh, "prompt": prompt[:200]})
    return {"id": tid, "model": body.model}

@router.get("/contents/generations/tasks/{task_id}")
async def poll(task_id: str, request: Request):
    app = request.app
    row = app.state.store.get(task_id)
    if row is None:
        raise HTTPException(404)
    entry = _registry(request).get_entry(row["model"])
    if row["status"] in ("queued", "running"):
        try:
            hist = await app.state.comfy.history(row["prompt_id"])
            st, vids, err = parse_history(hist)
            now = time.time()
            if st == "succeeded":
                f = vids[0]
                data = await app.state.comfy.fetch_output(f["filename"], f.get("subfolder", ""),
                                                          f.get("type", "output"))
                dest = Path(app.state.files_dir, f"{task_id}.mp4")
                dest.write_bytes(data)
                app.state.store.set_status(task_id, "succeeded",
                                           video_file=str(dest.relative_to(dest.parents[1])))
            elif st == "failed":
                app.state.store.set_status(task_id, "failed", error=err)
            elif now - row["created_at"] > entry.timeout_s:
                app.state.store.set_status(task_id, "failed", error="timeout")
            else:
                if st == "running" and row["status"] != "running":
                    app.state.store.set_status(task_id, "running")
                row = app.state.store.get(task_id)
        except Exception:
            pass    # 瞬时不可达：返回缓存状态
    row = app.state.store.get(task_id)
    resp = {"id": row["id"], "model": row["model"], "status": row["status"],
            "progress": None, "created_at": row["created_at"],
            "content": {}, "error": {"message": row["error"]} if row["error"] else None}
    if row["status"] == "succeeded":
        host = app.state.public_base  # 形如 http://127.0.0.1:8300
        resp["content"] = {"video_url": f"{host}/files/{row['id']}.mp4"}
    return resp

"""端到端冒烟：造首帧图 → 提交 → 轮询 → 下载 → ffprobe 断言。用法:
python scripts/smoke_local.py <model> <duration> <resolution> <aspect: 9:16|16:9>
"""
import base64, io, json, sys, time, urllib.request
from PIL import Image

BASE = "http://127.0.0.1:8300"

def make_first_frame(aspect: str) -> str:
    w, h = (720, 1280) if aspect == "9:16" else (1280, 720)
    img = Image.new("RGB", (w, h), (30 + w % 200, 90, 160))
    buf = io.BytesIO(); img.save(buf, "PNG")
    return "data:image/png;base64," + base64.b64encode(buf.getvalue()).decode()

def main():
    model, dur, res, aspect = sys.argv[1], int(sys.argv[2]), sys.argv[3], sys.argv[4]
    body = json.dumps({"model": model, "content": [
        {"type": "image_url", "image_url": {"url": make_first_frame(aspect)}},
        {"type": "text", "text": "slow cinematic dolly forward, subtle light drift"}],
        "duration": dur, "resolution": res}).encode()
    r = urllib.request.urlopen(urllib.request.Request(
        f"{BASE}/contents/generations/tasks", data=body,
        headers={"Content-Type": "application/json"}), timeout=60)
    tid = json.loads(r.read())["id"]
    print("task:", tid)
    t0 = time.time()
    while True:
        j = json.loads(urllib.request.urlopen(f"{BASE}/contents/generations/tasks/{tid}",
                                              timeout=30).read())
        print(f"[{time.time()-t0:6.0f}s] {j['status']}")
        if j["status"] in ("succeeded", "failed"):
            break
        time.sleep(15)
    if j["status"] != "succeeded":
        print("FAILED:", j.get("error")); sys.exit(1)
    mp4 = j["content"]["video_url"]
    out = f"smoke_{tid}.mp4"
    urllib.request.urlretrieve(mp4, out)
    print("saved:", out, "elapsed:", round(time.time() - t0), "s")

if __name__ == "__main__":
    main()
```
